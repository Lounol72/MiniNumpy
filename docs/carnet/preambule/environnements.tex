% Un seul compteur par chapitre pour tous les blocs numérotés :
% Définition 3.1, Exemple 3.2, Proposition 3.3...

\theoremstyle{definition}

\newtheorem{definition}{Définition}[chapter]
\newtheorem{example}[definition]{Exemple}

\theoremstyle{plain}

\newtheorem{theorem}[definition]{Théorème}
\newtheorem{proposition}[definition]{Proposition}
\newtheorem{lemma}[definition]{Lemme}
\newtheorem{corollary}[definition]{Corollaire}

\theoremstyle{remark}

\newtheorem{remark}[definition]{Remarque}

% Noms utilisés par \cref
\crefname{definition}{définition}{définitions}
\Crefname{definition}{Définition}{Définitions}
\crefname{example}{exemple}{exemples}
\Crefname{example}{Exemple}{Exemples}
\crefname{theorem}{théorème}{théorèmes}
\Crefname{theorem}{Théorème}{Théorèmes}
\crefname{proposition}{proposition}{propositions}
\Crefname{proposition}{Proposition}{Propositions}
\crefname{remark}{remarque}{remarques}
\Crefname{remark}{Remarque}{Remarques}

% Définitions et exemples : un trait vertical à gauche pour les repérer
\tcolorboxenvironment{definition}{
    blanker, breakable,
    borderline west={2pt}{0pt}{accent},
    left=10pt, before skip=8pt, after skip=8pt
}
\tcolorboxenvironment{example}{
    blanker, breakable,
    borderline west={2pt}{0pt}{discret!60},
    left=10pt, before skip=8pt, after skip=8pt
}
